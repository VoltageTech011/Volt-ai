require('./config'); const { default: makeWASocket, useMultiFileAuthState, DisconnectReason, fetchLatestWaWebVersion } = require('@whiskeysockets/baileys'); const { Boom } = require('@hapi/boom'); const pino = require('pino'); const express = require('express'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); const QRCode = require('qrcode'); const PORT = process.env.PORT || 3000; const SESSION_DIR = path.join(__dirname, 'session'); const PLUGIN_DIR = path.join(__dirname, 'plugins'); const app = express(); app.use(express.json()); app.use(express.urlencoded({ extended: false })); fs.mkdirSync(SESSION_DIR, { recursive: true }); fs.mkdirSync(PLUGIN_DIR, { recursive: true }); let sock = null; let connectionStatus = 'starting'; let pairingCode = null; let pairingCodeTime = 0; let reconnectTimer = null; let starting = false; const commandMap = new Map(); const hookPlugins = []; global.commandMap = commandMap; function normalizeJid(jid = '') { return String(jid).replace(/:\d+(?=@)/, ''); } function getNumberFromJid(jid = '') { return normalizeJid(jid).split('@')[0]; } function loadPlugins() { commandMap.clear(); hookPlugins.length = 0; if (!fs.existsSync(PLUGIN_DIR)) { fs.mkdirSync(PLUGIN_DIR, { recursive: true }); return; } const files = fs.readdirSync(PLUGIN_DIR) .filter(file => file.endsWith('.js')); for (const file of files) { try { const plugin = require(path.join(PLUGIN_DIR, file)); if (!plugin?.name) continue; if (typeof plugin.execute === 'function') { commandMap.set(plugin.name.toLowerCase(), plugin); if (Array.isArray(plugin.aliases)) { for (const alias of plugin.aliases) { commandMap.set(String(alias).toLowerCase(), plugin); } } } if (typeof plugin.onMessage === 'function') { hookPlugins.push(plugin); } console.log(`⚡ Plugin loaded: ${plugin.name}`); } catch (error) { console.error(`Plugin error [${file}]:`, error.message); } } console.log( `⚡ Voltage loaded ${commandMap.size} commands and ${hookPlugins.length} message hooks` ); } function checkOwner(sender = '') { const number = getNumberFromJid(sender); return global.owners.some(owner => { return getNumberFromJid(owner) === number; }); } function checkDev(sender = '') { const number = getNumberFromJid(sender); return global.dev.some(dev => { return getNumberFromJid(dev) === number; }); } function getMessageBody(message) { if (!message) return ''; if (message.conversation) { return message.conversation; } if (message.extendedTextMessage?.text) { return message.extendedTextMessage.text; } if (message.imageMessage?.caption) { return message.imageMessage.caption; } if (message.videoMessage?.caption) { return message.videoMessage.caption; } if (message.documentMessage?.caption) { return message.documentMessage.caption; } return ''; } async function serializeMessage(message) { const remoteJid = message.key?.remoteJid || ''; const isGroup = remoteJid.endsWith('@g.us'); const sender = message.key?.fromMe ? sock?.user?.id || '' : isGroup ? message.key?.participant || '' : remoteJid; const type = Object.keys(message.message || {})[0] || ''; const body = getMessageBody(message.message); return { key: message.key, id: message.key?.id || '', from: remoteJid, sender, senderNumber: getNumberFromJid(sender), pushName: message.pushName || getNumberFromJid(sender), isGroup, body, text: body, type, mtype: type, isOwner: checkOwner(sender), isDev: checkDev(sender), isMedia: [ 'imageMessage', 'videoMessage', 'audioMessage', 'documentMessage', 'stickerMessage' ].includes(type), reply: async (text, options = {}) => { return sock.sendMessage( remoteJid, { text: String(text), ...options }, { quoted: message } ); }, send: async content => { return sock.sendMessage(remoteJid, content); }, react: async emoji => { return sock.sendMessage(remoteJid, { react: { text: emoji, key: message.key } }); } }; } async function handleMessage(message) { if (!message?.message) return; const m = await serializeMessage(message); for (const plugin of hookPlugins) { if (message.key?.fromMe && !plugin.allowFromMe) { continue; } try { const blocked = await plugin.onMessage(sock, m); if (blocked === true) { return; } } catch (error) { console.error( `Hook error [${plugin.name}]:`, error.message ); } } if (!m.body) return; if (!m.body.startsWith(global.BOT_PREFIX)) { return; } const parts = m.body .slice(global.BOT_PREFIX.length) .trim() .split(/\s+/); const commandName = parts.shift()?.toLowerCase(); if (!commandName) return; const plugin = commandMap.get(commandName); if (!plugin) return; try { await plugin.execute(sock, m, parts); } catch (error) { console.error( `Command error [${commandName}]:`, error.message ); try { await m.reply(`Voltage encountered an error: ${error.message}`); } catch (_) {} } } async function startVoltage() { if (starting) return; if ( sock && connectionStatus === 'connected' ) { return; } starting = true; connectionStatus = 'connecting'; console.log('⚡ Starting Voltage...'); try { const { version } = await fetchLatestWaWebVersion(); const { state, saveCreds } = await useMultiFileAuthState(SESSION_DIR); sock = makeWASocket({ version, logger: pino({ level: 'silent' }), auth: state, printQRInTerminal: false, browser: [ 'Voltage', 'Chrome', '1.0.0' ], keepAliveIntervalMs: 10000, markOnlineOnConnect: true, syncFullHistory: false }); starting = false; sock.ev.on( 'creds.update', saveCreds ); sock.ev.on( 'connection.update', async update => { const { connection, lastDisconnect, qr } = update; if (qr) { try { pairingCode = null; global.voltageQR = await QRCode.toDataURL(qr); connectionStatus = 'waiting_for_qr'; console.log( '⚡ QR code generated.' ); } catch (error) { console.error( 'QR generation error:', error.message ); } } if (connection === 'connecting') { connectionStatus = 'connecting'; console.log( '⚡ Connecting to WhatsApp...' ); } if (connection === 'open') { connectionStatus = 'connected'; global.voltageQR = null; pairingCode = null; console.log( `⚡ Voltage connected as ${sock.user?.id || 'unknown'}` ); try { await sock.sendMessage( sock.user.id, { text: [ '⚡ VOLTAGE ONLINE', '', 'WhatsApp connection established.', '', `Prefix: ${global.BOT_PREFIX}`, `Commands: ${commandMap.size}`, '', 'Brain systems will be connected in later stages.' ].join('\n') } ); } catch (_) {} } if (connection === 'close') { connectionStatus = 'disconnected'; const statusCode = lastDisconnect?.error instanceof Boom ? lastDisconnect.error.output.statusCode : 0; const loggedOut = statusCode === DisconnectReason.loggedOut; console.log( `⚠️ Voltage disconnected. Code: ${statusCode}` ); sock = null; if (loggedOut) { console.log( '🔒 WhatsApp logged Voltage out.' ); try { fs.rmSync( SESSION_DIR, { recursive: true, force: true } ); fs.mkdirSync( SESSION_DIR, { recursive: true } ); } catch (_) {} connectionStatus = 'logged_out'; return; } if (reconnectTimer) { clearTimeout(reconnectTimer); } reconnectTimer = setTimeout(() => { reconnectTimer = null; startVoltage(); }, 5000); } } ); sock.ev.on( 'messages.upsert', async ({ messages }) => { for (const message of messages) { try { await handleMessage(message); } catch (error) { console.error( 'Message handler error:', error.message ); } } } ); } catch (error) { starting = false; sock = null; connectionStatus = 'error'; console.error( 'Voltage startup error:', error.message ); if (reconnectTimer) { clearTimeout(reconnectTimer); } reconnectTimer = setTimeout(() => { reconnectTimer = null; startVoltage(); }, 10000); } } app.get('/', (_req, res) => { res.json({ name: 'Voltage', status: connectionStatus, version: '1.0.0' }); }); app.get('/health', (_req, res) => { res.json({ status: 'ok', voltage: connectionStatus, uptime: process.uptime() }); }); app.get('/api/status', (_req, res) => { res.json({ name: 'Voltage', status: connectionStatus, connected: connectionStatus === 'connected', user: sock?.user?.id || null, prefix: global.BOT_PREFIX, commands: commandMap.size, qr: global.voltageQR || null, pairingCode: pairingCode && Date.now() - pairingCodeTime < 300000 ? pairingCode : null }); }); app.post('/api/pair', async (req, res) => { try { if (!sock) { return res.status(503).json({ error: 'Voltage is still starting.' }); } if (connectionStatus !== 'connecting') { return res.status(400).json({ error: `Voltage is currently ${connectionStatus}.` }); } const phone = String( req.body.phone || '' ).replace(/\D/g, ''); if (!phone) { return res.status(400).json({ error: 'Phone number is required.' }); } const code = await sock.requestPairingCode(phone); pairingCode = code; pairingCodeTime = Date.now(); res.json({ success: true, code }); } catch (error) { console.error( 'Pairing error:', error.message ); res.status(500).json({ error: error.message }); } }); app.get('/api/qr', (_req, res) => { res.json({ qr: global.voltageQR || null }); }); app.listen(PORT, () => { console.log( `⚡ Voltage server running on port ${PORT}` ); loadPlugins(); startVoltage(); }); process.on('SIGINT', async () => { console.log('⚡ Shutting down Voltage...'); try { sock?.end(); } catch (_) {} process.exit(0); }); process.on('SIGTERM', async () => { console.log('⚡ Shutting down Voltage...'); try { sock?.end(); } catch (_) {} process.exit(0); }); process.on( 'uncaughtException', error => { console.error( 'Uncaught exception:', error.message ); } ); process.on( 'unhandledRejection', error => { console.error( 'Unhandled rejection:', error?.message || error ); } );
